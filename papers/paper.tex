\begin{abstract}
Hadwiger's Conjecture (1943) asserts that every simple graph with chromatic number $\chi(G) = k$ contains the complete graph $K_k$ as a minor. In this paper, we present three interconnected contributions:
(1) Universal closed-form theorems and analytic proofs for the Hadwiger numbers $h(G)$ of multiple infinite graph families, including the Universal Apex Theorem ($h(G \star K_1) = h(G) + 1$), the Complete Multipartite Theorem ($h(K_{n_1, \dots, n_k}) = \min(1 + \sum_{i=1}^{k-1} n_i, \lfloor (n + k)/2 \rfloor)$), the Tur\'an Hadwiger Theorem ($h(T(n, r)) = \lfloor (n + r)/2 \rfloor$), and complete classifications for 2D grids $P_a \square P_b$, cylinders $C_a \square P_b$, and hypercubes $Q_d$;
(2) Exhaustive computational atlases surveying all 13,598 non-isomorphic graphs up to $n=8$ (evaluating all 12,346 graphs at $n=8$ in 60.85s), establishing 0 violations of Hadwiger's and the Weak Hadwiger Conjectures, and isolating all 34 tight extremal graphs satisfying $h \cdot \alpha = 8$; and
(3) Machine-checked kernel verification in Lean 4 with 0 axioms, 0 \texttt{sorry}, and 0 \texttt{native\_decide}, providing integer branch-set certificates for all canonical instances.
\end{abstract}

\section{Introduction}
Let $G = (V, E)$ be a finite, simple, undirected graph. The Hadwiger number $h(G)$ is the maximum order $t$ such that the complete graph $K_t$ is a minor of $G$. In 1943, Hugo Hadwiger conjectured that $h(G) \ge \chi(G)$ for all simple graphs.

\section{Main Theorems}

\begin{theorem}[Universal Apex Theorem]
For any simple graph $G$, $h(G \star K_1) = h(G) + 1$, and for any $k \ge 1$, $h(G \star K_k) = h(G) + k$.
\end{theorem}

\begin{theorem}[Complete Multipartite Hadwiger Theorem]
For any complete multipartite graph $K_{n_1, \dots, n_k}$ with parts in non-decreasing order $1 \le n_1 \le \dots \le n_k$ and $n = \sum_{i=1}^k n_i$:
\[
h(K_{n_1, \dots, n_k}) = \min\left(1 + \sum_{i=1}^{k-1} n_i, \left\lfloor \frac{n + k}{2} \right\rfloor\right)
\]
\end{theorem}

\begin{theorem}[Tur\'an Hadwiger Theorem]
For all integers $n \ge r \ge 2$, the balanced complete $r$-partite Tur\'an graph $T(n, r)$ satisfies:
\[
h(T(n, r)) = \left\lfloor \frac{n + r}{2} \right\rfloor
\]
\end{theorem}

\begin{theorem}[Cartesian Grid \& Product Theorem]
\begin{enumerate}
    \item 2D Grids $P_a \square P_b$: $h(P_1 \square P_1) = 1$; for $\min(a, b)=1, \max(a, b)\ge 2$, $h = 2$; for $\min(a, b)=2, \max(a, b)\ge 2$, $h = 3$; for $\min(a, b)\ge 3$, $h = 4$.
    \item Cylinders $C_a \square P_b$ ($a \ge 3$): $h(C_a \square P_1) = 3$; for $b \ge 2$, $h(C_a \square P_b) = 4$.
    \item Hypercubes: $h(Q_0) = 1, h(Q_1) = 2, h(Q_2) = 3, h(Q_3) = 4$.
\end{enumerate}
\end{theorem}
